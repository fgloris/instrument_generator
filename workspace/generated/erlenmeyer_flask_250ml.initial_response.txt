<BLENDER_SCRIPT>
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Generate a physically coherent 250 mL Erlenmeyer flask in Blender 5.2."""

from __future__ import annotations

import importlib
import os
import sys
from pathlib import Path

import bpy
from mathutils import Vector


SCRIPT_DIR = Path(__file__).resolve().parent
TOOLKIT_DIR = SCRIPT_DIR.parent / "toolkit"
if str(TOOLKIT_DIR) not in sys.path:
    sys.path.insert(0, str(TOOLKIT_DIR))

import lab_blender_toolkit as lab

importlib.reload(lab)


NAME = "Erlenmeyer_Flask_250mL"
INSTRUMENT_ID = "erlenmeyer_flask_250ml"
NOMINAL_VOLUME_ML = 250
HEIGHT_MM = 145.0
BASE_DIAMETER_MM = 82.0
NECK_OUTER_DIAMETER_MM = 30.0
NOMINAL_WALL_THICKNESS_MM = 2.0
RADIAL_SEGMENTS = 192

OUTPUT_DIR = Path(
    os.getenv("LAB_ASSET_OUTPUT_DIR", str(SCRIPT_DIR / "output" / NAME))
).resolve()
RENDER_ENGINE = os.getenv("LAB_RENDER_ENGINE", "BLENDER_EEVEE")
RESOLUTION = int(os.getenv("LAB_RENDER_RESOLUTION", "768"))

OUTER_PROFILE = lab.smooth_profile_from_mm(
    [
        (40.0, 0.0),
        (41.0, 2.0),
        (41.0, 7.0),
        (39.5, 18.0),
        (35.0, 55.0),
        (29.0, 82.0),
        (22.0, 99.0),
        (16.0, 108.0),
        (15.0, 113.0),
        (15.0, 138.0),
        (16.8, 140.5),
        (17.0, 143.5),
        (16.0, 145.0),
    ],
    samples_per_segment=10,
    sharp_indices={0, 1, 8, 9, 10, 11, 12},
)

INNER_PROFILE = lab.smooth_profile_from_mm(
    [
        (38.0, 4.0),
        (39.0, 7.0),
        (37.5, 18.0),
        (33.0, 55.0),
        (27.0, 82.0),
        (20.0, 99.0),
        (13.0, 108.0),
        (13.0, 113.0),
        (13.0, 138.0),
        (14.5, 140.5),
        (14.7, 143.5),
        (13.0, 145.0),
    ],
    samples_per_segment=10,
    sharp_indices={0, 7, 8, 9, 10, 11},
)

GRADUATION_STYLE = lab.GraduationStyle(
    end_angle_deg=-35.0,
    major_length=lab.mm(15.0),
    minor_length=lab.mm(8.0),
    line_radius=lab.mm(0.24),
    surface_clearance=lab.mm(0.008),
    label_tangent_offset=lab.mm(4.8),
    text_size=lab.mm(4.0),
    text_extrude=lab.mm(0.010),
    text_outline=lab.mm(0.005),
    text_surface_clearance=lab.mm(0.006),
    text_curve_resolution=12,
    text_subdivision_cuts=1,
    top_clearance=lab.mm(18.0),
    arc_samples=28,
)

UNIT_LABEL_STYLE = lab.GraduationStyle(
    end_angle_deg=-35.0,
    major_length=lab.mm(15.0),
    minor_length=lab.mm(8.0),
    line_radius=lab.mm(0.24),
    surface_clearance=lab.mm(0.008),
    label_tangent_offset=lab.mm(-8.0),
    text_size=lab.mm(3.5),
    text_extrude=lab.mm(0.010),
    text_outline=lab.mm(0.004),
    text_surface_clearance=lab.mm(0.006),
    text_curve_resolution=12,
    text_subdivision_cuts=1,
    top_clearance=lab.mm(18.0),
    arc_samples=28,
)

VIEWS = [
    lab.RenderView(
        name="graduations_front",
        azimuth_deg=-50.0,
        elevation_deg=17.0,
        distance=0.305,
        lens_mm=62.0,
    ),
    lab.RenderView(
        name="mouth_and_shoulder",
        azimuth_deg=35.0,
        elevation_deg=31.0,
        distance=0.315,
        lens_mm=64.0,
    ),
    lab.RenderView(
        name="rear_three_quarter",
        azimuth_deg=142.0,
        elevation_deg=20.0,
        distance=0.310,
        lens_mm=62.0,
    ),
    lab.RenderView(
        name="low_base_diagnostic",
        azimuth_deg=-58.0,
        elevation_deg=9.0,
        distance=0.300,
        lens_mm=60.0,
    ),
]


def validate_geometry() -> float:
    capacity_ml = lab.profile_capacity_ml(INNER_PROFILE)
    if capacity_ml < NOMINAL_VOLUME_ML:
        raise RuntimeError(
            f"Inner profile capacity is {capacity_ml:.2f} mL, below the "
            f"required {NOMINAL_VOLUME_ML} mL."
        )

    nominal_mark_z = lab.height_for_volume_ml(
        INNER_PROFILE,
        NOMINAL_VOLUME_ML,
    )
    shoulder_start_z = lab.mm(82.0)
    if nominal_mark_z >= shoulder_start_z:
        raise RuntimeError(
            "The nominal 250 mL graduation must remain below the shoulder."
        )
    return capacity_ml


def build_asset() -> bpy.types.Object:
    capacity_ml = validate_geometry()

    lab.clear_scene()
    lab.configure_scene(
        resolution=RESOLUTION,
        transparent_background=False,
        render_engine=RENDER_ENGINE,
    )

    asset_collection = lab.create_collection("ASSET_Erlenmeyer_Flask")
    marking_collection = lab.create_collection("MARKINGS_Graduations")
    studio_collection = lab.create_collection("STUDIO_Diagnostic")

    glass_material = lab.create_borosilicate_glass_material(
        name="MAT_Clear_Borosilicate_Glass",
        tint=(1.0, 1.0, 1.0, 1.0),
        roughness=0.014,
        ior=1.47,
        absorption_density=0.030,
    )
    marking_material = lab.create_marking_material(
        name="MAT_White_Approximate_Graduations"
    )

    flask = lab.create_hollow_revolved_mesh(
        name=NAME,
        outer_profile=OUTER_PROFILE,
        inner_profile=INNER_PROFILE,
        collection=asset_collection,
        material=glass_material,
        radial_segments=RADIAL_SEGMENTS,
    )

    nominal_mark_z_mm = (
        lab.height_for_volume_ml(INNER_PROFILE, NOMINAL_VOLUME_ML) * 1000.0
    )
    lab.assign_metadata(
        flask,
        {
            "asset_id": INSTRUMENT_ID,
            "instrument_type": "erlenmeyer_flask",
            "name_en": "250 mL Erlenmeyer flask",
            "category": "laboratory_glassware",
            "material_type": "clear_borosilicate_glass",
            "nominal_volume_ml": NOMINAL_VOLUME_ML,
            "calibration_class": "approximate",
            "calculated_inner_capacity_ml": capacity_ml,
            "nominal_mark_height_mm": nominal_mark_z_mm,
            "outer_height_mm": HEIGHT_MM,
            "base_diameter_mm": BASE_DIAMETER_MM,
            "neck_outer_diameter_mm": NECK_OUTER_DIAMETER_MM,
            "nominal_wall_thickness_mm": NOMINAL_WALL_THICKNESS_MM,
            "bottom_thickness_mm": 4.0,
            "mouth_open": True,
            "has_liquid": False,
            "has_stopper": False,
        },
    )

    lab.add_volume_graduations(
        inner_profile=INNER_PROFILE,
        outer_profile=OUTER_PROFILE,
        nominal_volume_ml=NOMINAL_VOLUME_ML,
        minor_interval_ml=25,
        major_interval_ml=50,
        style=GRADUATION_STYLE,
        material=marking_material,
        collection=marking_collection,
    )

    unit_label = lab.create_conformal_surface_label(
        name="Label_Approximate_mL",
        text="APPROX. mL",
        z=lab.mm(91.0),
        outer_profile=OUTER_PROFILE,
        style=UNIT_LABEL_STYLE,
        material=marking_material,
        collection=marking_collection,
    )
    unit_label["marking_type"] = "approximate_volume_unit_label"

    lab.create_grid_floor(
        collection=studio_collection,
        size=0.55,
        z=lab.mm(-0.15),
        spacing=lab.mm(10.0),
        minor_width=lab.mm(0.22),
        major_width=lab.mm(0.55),
        major_every=5,
    )

    lab.enable_freestyle_outline(
        thickness_px=1.25,
        color=(0.015, 0.015, 0.018, 1.0),
        include_open_borders=True,
        include_creases=False,
    )

    camera = lab.create_camera(
        collection=studio_collection,
        name="Diagnostic_Camera",
    )
    target = Vector((0.0, 0.0, lab.mm(72.5)))
    lab.setup_glass_product_lighting(studio_collection, target)

    first_view = VIEWS[0]
    camera.location = lab.orbit_location(target, first_view)
    camera.data.lens = first_view.lens_mm
    lab.look_at(camera, target)
    lab.select_only(flask)

    print("=" * 72)
    print(f"Generated: {NAME}")
    print(f"Integrated inner capacity: {capacity_ml:.2f} mL")
    print(f"250 mL mark height: {nominal_mark_z_mm:.2f} mm")
    print(f"Render engine: {RENDER_ENGINE}")
    print(f"Render resolution: {RESOLUTION} x {RESOLUTION}")
    print(f"Output directory: {OUTPUT_DIR}")
    print("=" * 72)

    lab.render_views(
        camera=camera,
        target=target,
        views=VIEWS,
        output_directory=OUTPUT_DIR,
        filename_prefix=NAME,
    )
    lab.save_blend(OUTPUT_DIR / f"{NAME}.blend")
    return flask


if __name__ == "__main__":
    build_asset()
</BLENDER_SCRIPT>
