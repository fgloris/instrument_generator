<BLENDER_SCRIPT>
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Generate a physically coherent 250 mL Erlenmeyer flask in Blender 5.2."""

from __future__ import annotations

import importlib
import os
import sys
from pathlib import Path

import bpy
from mathutils import Vector


SCRIPT_DIR = Path(__file__).resolve().parent
TOOLKIT_DIR = SCRIPT_DIR.parent / "toolkit"
if str(TOOLKIT_DIR) not in sys.path:
    sys.path.insert(0, str(TOOLKIT_DIR))

import lab_blender_toolkit as lab

importlib.reload(lab)


NAME = "Erlenmeyer_Flask_250mL"
NOMINAL_VOLUME_ML = 250
HEIGHT_MM = 145.0
BASE_DIAMETER_MM = 82.0
NECK_OUTER_DIAMETER_MM = 30.0
NOMINAL_WALL_THICKNESS_MM = 2.0
SHOULDER_START_MM = 84.0
RADIAL_SEGMENTS = 192

OUTPUT_DIR = Path(
    os.getenv(
        "LAB_ASSET_OUTPUT_DIR",
        str(SCRIPT_DIR / "output" / NAME),
    )
).resolve()
RENDER_ENGINE = os.getenv("LAB_RENDER_ENGINE", "BLENDER_EEVEE")
RESOLUTION = int(os.getenv("LAB_RENDER_RESOLUTION", "768"))

if RESOLUTION <= 0:
    raise ValueError("LAB_RENDER_RESOLUTION must be a positive integer.")

OUTER_PROFILE = lab.smooth_profile_from_mm(
    [
        (40.5, 0.0),
        (41.0, 2.5),
        (40.5, 7.0),
        (36.9, 70.0),
        (35.5, 84.0),
        (30.0, 99.0),
        (22.0, 112.0),
        (16.0, 121.0),
        (15.0, 126.0),
        (15.0, 139.0),
        (17.0, 141.5),
        (17.2, 143.5),
        (16.5, 145.0),
    ],
    samples_per_segment=10,
    sharp_indices={0, 1, 12},
)

INNER_PROFILE = lab.smooth_profile_from_mm(
    [
        (38.5, 4.5),
        (38.5, 7.0),
        (34.9, 70.0),
        (33.5, 84.0),
        (28.0, 99.0),
        (20.0, 112.0),
        (14.0, 121.0),
        (13.0, 126.0),
        (13.0, 139.0),
        (14.0, 141.5),
        (14.2, 143.5),
        (13.2, 145.0),
    ],
    samples_per_segment=10,
    sharp_indices={0, 11},
)

GRADUATION_STYLE = lab.GraduationStyle(
    end_angle_deg=-34.0,
    major_length=lab.mm(15.0),
    minor_length=lab.mm(8.0),
    line_radius=lab.mm(0.25),
    surface_clearance=lab.mm(0.015),
    label_tangent_offset=lab.mm(4.8),
    text_size=lab.mm(3.9),
    text_extrude=lab.mm(0.014),
    text_outline=lab.mm(0.005),
    text_surface_clearance=lab.mm(0.010),
    text_curve_resolution=12,
    text_subdivision_cuts=2,
    top_clearance=lab.mm(5.0),
    arc_samples=28,
)

AUXILIARY_LABEL_STYLE = lab.GraduationStyle(
    end_angle_deg=-34.0,
    major_length=lab.mm(15.0),
    minor_length=lab.mm(8.0),
    line_radius=lab.mm(0.25),
    surface_clearance=lab.mm(0.015),
    label_tangent_offset=lab.mm(4.8),
    text_size=lab.mm(3.2),
    text_extrude=lab.mm(0.014),
    text_outline=lab.mm(0.004),
    text_surface_clearance=lab.mm(0.010),
    text_curve_resolution=12,
    text_subdivision_cuts=2,
    top_clearance=lab.mm(5.0),
    arc_samples=28,
)

VIEWS = [
    lab.RenderView(
        "graduations_front_three_quarter",
        -50.0,
        17.0,
        0.315,
        62.0,
    ),
    lab.RenderView(
        "shoulder_and_neck",
        30.0,
        22.0,
        0.320,
        64.0,
    ),
    lab.RenderView(
        "mouth_high_angle",
        -120.0,
        34.0,
        0.330,
        64.0,
    ),
    lab.RenderView(
        "base_and_profile",
        142.0,
        12.0,
        0.310,
        62.0,
    ),
]


def validate_geometry() -> tuple[float, float]:
    capacity_ml = lab.profile_capacity_ml(INNER_PROFILE)
    if capacity_ml < NOMINAL_VOLUME_ML:
        raise RuntimeError(
            f"Inner profile capacity is {capacity_ml:.2f} mL; "
            f"at least {NOMINAL_VOLUME_ML} mL is required."
        )

    nominal_mark_z = lab.height_for_volume_ml(
        INNER_PROFILE,
        NOMINAL_VOLUME_ML,
    )
    if nominal_mark_z >= lab.mm(SHOULDER_START_MM):
        raise RuntimeError(
            "The calculated 250 mL graduation is not below the shoulder."
        )

    return capacity_ml, nominal_mark_z


def build_asset() -> bpy.types.Object:
    capacity_ml, nominal_mark_z = validate_geometry()

    lab.clear_scene()
    lab.configure_scene(
        resolution=RESOLUTION,
        transparent_background=False,
        render_engine=RENDER_ENGINE,
    )

    asset_collection = lab.create_collection("ASSET_ErlenmeyerFlask")
    marking_collection = lab.create_collection("ASSET_Graduations")
    studio_collection = lab.create_collection("STUDIO")

    glass_material = lab.create_borosilicate_glass_material(
        name="MAT_Clear_Borosilicate_Glass",
        tint=(1.0, 1.0, 1.0, 1.0),
        roughness=0.014,
        ior=1.47,
        absorption_density=0.030,
    )
    marking_material = lab.create_marking_material(
        name="MAT_White_Approximate_Graduations"
    )

    flask = lab.create_hollow_revolved_mesh(
        name=NAME,
        outer_profile=OUTER_PROFILE,
        inner_profile=INNER_PROFILE,
        collection=asset_collection,
        material=glass_material,
        radial_segments=RADIAL_SEGMENTS,
    )

    lab.assign_metadata(
        flask,
        {
            "asset_id": "erlenmeyer_flask_250ml",
            "name_en": "250 mL Erlenmeyer flask",
            "name_zh": "250 mL 锥形瓶",
            "instrument_type": "erlenmeyer_flask",
            "category": "laboratory_glassware",
            "material_type": "clear_borosilicate_glass",
            "nominal_volume_ml": NOMINAL_VOLUME_ML,
            "calculated_inner_capacity_ml": round(capacity_ml, 3),
            "nominal_mark_height_mm": round(nominal_mark_z * 1000.0, 3),
            "outer_height_mm": HEIGHT_MM,
            "maximum_base_diameter_mm": BASE_DIAMETER_MM,
            "neck_outer_diameter_mm": NECK_OUTER_DIAMETER_MM,
            "nominal_wall_thickness_mm": NOMINAL_WALL_THICKNESS_MM,
            "bottom_thickness_mm": 4.5,
            "graduation_type": "approximate",
            "graduation_minor_interval_ml": 25,
            "graduation_major_interval_ml": 50,
            "mouth_type": "open_rolled_rim",
        },
    )

    lab.add_volume_graduations(
        inner_profile=INNER_PROFILE,
        outer_profile=OUTER_PROFILE,
        nominal_volume_ml=NOMINAL_VOLUME_ML,
        minor_interval_ml=25,
        major_interval_ml=50,
        style=GRADUATION_STYLE,
        material=marking_material,
        collection=marking_collection,
    )

    auxiliary_label_z = nominal_mark_z + lab.mm(7.0)
    lab.create_conformal_surface_label(
        name="Label_mL_APPROX",
        text="mL APPROX",
        z=auxiliary_label_z,
        outer_profile=OUTER_PROFILE,
        style=AUXILIARY_LABEL_STYLE,
        material=marking_material,
        collection=marking_collection,
    )

    lab.create_grid_floor(
        collection=studio_collection,
        size=0.55,
        z=lab.mm(-0.15),
        spacing=lab.mm(10.0),
        minor_width=lab.mm(0.22),
        major_width=lab.mm(0.55),
        major_every=5,
    )

    camera = lab.create_camera(
        collection=studio_collection,
        name="Erlenmeyer_Diagnostic_Camera",
    )
    target = Vector((0.0, 0.0, lab.mm(72.0)))
    lab.setup_glass_product_lighting(studio_collection, target)

    camera.location = lab.orbit_location(target, VIEWS[0])
    camera.data.lens = VIEWS[0].lens_mm
    lab.look_at(camera, target)
    lab.select_only(flask)

    print("=" * 72)
    print(f"Generated: {NAME}")
    print(f"Integrated inner capacity: {capacity_ml:.2f} mL")
    print(f"Calculated 250 mL mark height: {nominal_mark_z * 1000.0:.2f} mm")
    print(f"Render engine: {RENDER_ENGINE}")
    print(f"Render resolution: {RESOLUTION} x {RESOLUTION}")
    print(f"Output directory: {OUTPUT_DIR}")
    print("=" * 72)

    lab.render_views(
        camera=camera,
        target=target,
        views=VIEWS,
        output_directory=OUTPUT_DIR,
        filename_prefix=NAME,
    )
    lab.save_blend(OUTPUT_DIR / f"{NAME}.blend")

    return flask


if __name__ == "__main__":
    build_asset()
</BLENDER_SCRIPT>
<SUMMARY>
Generates a dimensionally explicit 250 mL narrow-neck Erlenmeyer flask with a closed thick-bottomed hollow glass mesh, smooth conical shoulder, cylindrical neck, open thickened rim, and volume-integrated 25 mL graduations with 50 mL labels. The script validates capacity and 250 mL mark placement, adds metadata and a studio grid, renders four diagnostic views, and saves the Blender file under the configured output directory.
